def ensure_datetime_index(df: pd.DataFrame):
    if df is None or df.empty: return df
    df = df.copy()
    if not isinstance(df.index, pd.DatetimeIndex): 
        df.index = pd.to_datetime(df.index)
    return df

def render_tv_chart(df: pd.DataFrame, plan: dict):
    """TradingView Lightweight Chart Implementation for Pro-Level Dashboard."""
    chart_data = df.reset_index()
    chart_data.columns = [c.lower() for c in chart_data.columns]
    if 'date' in chart_data.columns: chart_data = chart_data.rename(columns={'date': 'time'})
    elif 'index' in chart_data.columns: chart_data = chart_data.rename(columns={'index': 'time'})
    chart_data['time'] = chart_data['time'].astype(str)
    
    candles = chart_data[['time', 'open', 'high', 'low', 'close']].to_dict('records')
    volume = chart_data[['time', 'volume']].rename(columns={'volume': 'value'}).to_dict('records')
    
    chart_options = {
        "height": 600,
        "layout": {"background": {"type": 'solid', "color": '#0B0E14'}, "textColor": '#d1d4dc'},
        "grid": {"vertLines": {"color": '#1f2937'}, "horzLines": {"color": '#1f2937'}},
        "priceScale": {"borderColor": '#2b2b43'},
        "timeScale": {"borderColor": '#2b2b43'}
    }
    
    series = [
        {"type": 'Candlestick', "data": candles, "options": {"upColor": '#26a69a', "downColor": '#ef5350'}},
        {"type": 'Histogram', "data": volume, "options": {"color": 'rgba(38, 166, 154, 0.3)', "priceFormat": {"type": 'volume'}, "priceScaleId": ""}}
    ]
    renderLightweightCharts(charts=[{"chart": chart_options, "series": series}])

def render_advanced_theory_chart(df: pd.DataFrame, plan: dict, active_patterns: list, show_ichimoku: bool, show_fibs: bool, osc_selection: list):
    """High-Fidelity Plotly Chart for Geometric Analysis (Elliott & Harmonics)."""
    df = ensure_datetime_index(df)
    
    if show_ichimoku:
        df['tenkan'] = (df['high'].rolling(9).max() + df['low'].rolling(9).min()) / 2
        df['kijun'] = (df['high'].rolling(26).max() + df['low'].rolling(26).min()) / 2

    rows = 1 + len(osc_selection)
    row_heights = [0.7] + [0.3 / len(osc_selection)] * len(osc_selection) if osc_selection else [1.0]
    fig = make_subplots(rows=rows, cols=1, shared_xaxes=True, vertical_spacing=0.02, row_heights=row_heights)
    
    # Candlestick
    fig.add_trace(go.Candlestick(x=df.index, open=df['open'], high=df['high'], low=df['low'], close=df['close'], name="Price"), row=1, col=1)

    # Elliott Wave (High-Fidelity Drawing)
    elliott = plan.get("elliott", {})
    if "wave_coords" in elliott:
        coords = elliott["wave_coords"]
        fig.add_trace(go.Scatter(
            x=[c['x'] for c in coords], y=[c['y'] for c in coords],
            mode='lines+markers+text', text=["1", "2", "3"], textposition="top center",
            line=dict(color='#facc15', width=3, dash='dash'), name="Elliott Path"
        ), row=1, col=1)

    # Harmonics
    for p in plan.get("harmonics", []):
        if p.get('name') in active_patterns:
            pts = p.get('points', [])
            if len(pts) >= 5:
                p_x = [df.index[min(int(pt[0]), len(df)-1)] for pt in pts]
                p_y = [pt[1] for pt in pts]
                fig.add_trace(go.Scatter(x=[p_x[0], p_x[1], p_x[2], p_x[3], p_x[4], p_x[2], p_x[0]], 
                                         y=[p_y[0], p_y[1], p_y[2], p_y[3], p_y[4], p_y[2], p_y[0]], 
                                         fill="toself", fillcolor='rgba(59, 130, 246, 0.15)', 
                                         line=dict(color='#3b82f6', width=2), name=p.get('name')), row=1, col=1)

    if show_ichimoku:
        fig.add_trace(go.Scatter(x=df.index, y=df['tenkan'], line=dict(color='blue', width=1), name='Tenkan'), row=1, col=1)
        fig.add_trace(go.Scatter(x=df.index, y=df['kijun'], line=dict(color='red', width=1), name='Kijun'), row=1, col=1)

    if show_fibs and "fib_levels" in plan:
        for name, price in plan["fib_levels"].items():
            fig.add_hline(y=price, line_dash="dot", line_color="gray", annotation_text=name, row=1, col=1)

    curr_row = 2
    if "RSI" in osc_selection and 'rsi' in df.columns:
        fig.add_trace(go.Scatter(x=df.index, y=df['rsi'], line=dict(color='#9333ea'), name="RSI"), row=curr_row, col=1)
        curr_row += 1
    if "MACD" in osc_selection and 'macd' in df.columns:
        fig.add_trace(go.Scatter(x=df.index, y=df['macd'], line=dict(color='#3b82f6'), name="MACD"), row=curr_row, col=1)
        fig.add_trace(go.Scatter(x=df.index, y=df['macds'], line=dict(color='#facc15'), name="Signal"), row=curr_row, col=1)
        fig.add_trace(go.Bar(x=df.index, y=df['macdh'], marker_color='rgba(128,128,128,0.5)', name="Histogram"), row=curr_row, col=1)
        curr_row += 1
    if "Volume" in osc_selection:
        fig.add_trace(go.Bar(x=df.index, y=df['volume'], name="Volume", marker_color='gray'), row=curr_row, col=1)

    fig.update_layout(template="plotly_dark", height=850, xaxis_rangeslider_visible=False, margin=dict(l=10, r=10, t=10, b=10))
    st.plotly_chart(fig, width='stretch')
